# Stellar dependencies — DIRECT dependencies only, pinned.
#
# This file lists only packages we deliberately chose. Transitive deps
# (anyio, httpx, pydantic, ...) are resolved by the installer and are not
# listed here, so that reading this file tells you what the project
# actually uses rather than what got dragged in.
#
# Install:  uv pip sync requirements.lock   (exact versions, see the end)
#
# Each block is added in the phase that first needs it. If you are on an
# early phase, later blocks simply will not exist yet.

# --- Phase 0: talk to the model, talk to Redis, read config ------------
# Google's current Gemini SDK. Note: NOT google-generativeai, which is the
# older deprecated package. This one builds function-calling schemas
# directly from Python type hints and docstrings — the feature phase 4
# depends on.
google-genai==2.23.0

# Redis client. Carries SSE stream buffers, cancellation pubsub and
# rate-limit block state across Gunicorn worker processes.
redis==8.1.0

# Loads keys.env into os.environ at startup.
python-dotenv==1.2.3

# --- Phase 1: web layer ------------------------------------------------
# Brings in Werkzeug (WSGI + password hashing), Jinja2 (templates),
# itsdangerous (signed session cookies) and click (the flask CLI).
# Password hashing uses werkzeug.security, so no separate bcrypt dep.
Flask==3.1.3

# --- Phase 4: agent tools ----------------------------------------------
# HTTP for fetch_url and the Tavily search API. Already present as a
# transitive dependency of google-genai, but declared here because this
# project imports it directly - a transitive dep can vanish in a version bump.
requests==2.34.2

# HTML to readable text for fetch_url. Used with the stdlib html.parser,
# so lxml is not required.
beautifulsoup4==4.15.0

# --- Phase 5: the sandbox ----------------------------------------------
# Docker Engine API client, for creating and exec-ing into lab containers.
# Pulls pywin32 on Windows to speak to the named pipe the daemon listens on.
docker==7.2.0

# The IANA timezone database, for get_current_time.
# Linux and macOS ship one system-wide and stdlib zoneinfo finds it there;
# Windows does not, so without this every lookup raises ZoneInfoNotFoundError
# and a perfectly valid name like Asia/Kolkata is rejected.
tzdata==2026.4

# --- Phase 8: the rest of the tool suite --------------------------------
# Writes .pptx files for make_presentation: real, editable slides rather
# than pictures of slides. Pulls in Pillow and lxml.
python-pptx==1.0.2

# Renders the Markdown body of send_self_email as the HTML alternative,
# so a report reads in a mail client the way it read in the chat.
markdown==3.10.3

# --- Phase 10: generative UI and games ---------------------------------
# Move generation, legality and FEN. The reason the agent cannot play an
# illegal move any more: the legal list is generated from the position
# rather than recalled from the conversation.
chess==1.11.2

# --- Phase 11: terminals ---------------------------------------------
# The SSH server behind ssh_gateway.py. Imported directly, so declared
# directly: without it a fresh install cannot start the gateway at all.
paramiko==5.0.0

# --- Google sign-in -----------------------------------------------------
# Verifies Firebase ID tokens. google-genai already depends on it, and
# that is exactly why the pin must agree with it: an exact pin below the
# range google-genai 2.23 requires (>=2.56) made this whole file
# uninstallable. A range, with the exact version recorded in the lock file.
google-auth>=2.56,<3

# --- Imported directly, previously only arriving by accident -----------
# PIL is imported for image work; it used to arrive only because
# python-pptx depends on it.
pillow==12.3.0

# --- Production server (Linux only; Gunicorn does not run on Windows) ---
gunicorn==23.0.0; sys_platform != "win32"

# The exact version of every package, direct and transitive, for Windows
# and Linux alike, is in requirements.lock. Install from that:
#     uv pip sync requirements.lock
# and regenerate it after changing this file:
#     uv pip compile requirements.txt --universal --python-version 3.12 -o requirements.lock
